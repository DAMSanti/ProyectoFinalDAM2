% Definir estilos de notación Chen
\tikzset{
    entity/.style={
        rectangle,
        draw=black,
        thick,
        minimum width=3cm,
        minimum height=1.5cm,
        text centered,
        font=\bfseries\small
    },
    weakentity/.style={
        rectangle,
        draw=black,
        thick,
        double,
        minimum width=2.5cm,
        minimum height=1.2cm,
        text centered,
        font=\bfseries
    },
    relationship/.style={
        diamond,
        draw=black,
        thick,
        minimum width=2.5cm,
        minimum height=1.5cm,
        text centered,
        font=\footnotesize,
        aspect=2
    },
    attribute/.style={
        ellipse,
        draw=black,
        minimum width=2cm,
        minimum height=0.9cm,
        text centered,
        font=\tiny
    },
    keyattribute/.style={
        ellipse,
        draw=black,
        thick,
        minimum width=2cm,
        minimum height=0.9cm,
        text centered,
        font=\tiny\bfseries,
        text decoration={underline}
    },
    multiattribute/.style={
        ellipse,
        draw=black,
        double,
        minimum width=2cm,
        minimum height=0.9cm,
        text centered,
        font=\tiny
    },
    derivedattribute/.style={
        ellipse,
        draw=black,
        dashed,
        minimum width=1.8cm,
        minimum height=0.8cm,
        text centered,
        font=\small
    }
}
